    x : float = 0.7
    y : float = 0.0

### abs_min_max_float
# expect: ok
    x : float = -1.5
    y : float = 0.0
    y = abs(x) + min(x, 2.0) * max(0.5, x)
    print('%.12f\n' % y)

### abs_min_max_int
# expect: ok
    k : int = -3
    j : int = 0
    j = abs(k) + min(k, 2) * max(4, k)
    print('%i\n' % j)

### dimv_repeated
# expect: ok
@sizes : dimv = [[n, 1]] * (n + 1)
@class C:
@    def __init__(self) -> None:
@        self.L : List = plist(pmat, sizes)
    c : C = C()
    for k in range(n + 1):
        c.L[k][0, 0] = 1.5
    pmat_print(c.L[n])

### list_element_element
# expect: ok
@sizes : dimv = [[n, n]] * 2
@class C:
@    def __init__(self) -> None:
@        self.L : List = plist(pmat, sizes)
    c : C = C()
    x : float = 0.0
    c.L[1][0, 1] = 2.5
    c.L[0][1, 1] = c.L[1][0, 1] * 2.0
    x = c.L[0][1, 1] + c.L[1][0, 1]
    print('%f\n' % x)

### self_method_calls
