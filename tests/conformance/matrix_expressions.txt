### cholesky_and_cho_solve

    for i in range(n):
        A[i, i] = 4.0
        A[0, i] = 1.0
        B[i, 0] = 1.0
        B[i, 1] = 2.0

    pmat_print(X)
    L = cholesky(A @ A.T + M)
    pmat_print(L)
    M = cho_solve(L, M).T
    pmat_print(M)
